if(SIR_WITH_D3D9)
    if(NOT WIN32)
        message(FATAL_ERROR "SIR_WITH_D3D9 requires Windows and the Windows SDK")
    endif()
    include(CheckIncludeFileCXX)
    check_include_file_cxx("d3d9.h" SIR_D3D9_HEADER_FOUND)
    if(NOT SIR_D3D9_HEADER_FOUND)
        message(FATAL_ERROR "SIR_WITH_D3D9 requires Windows SDK d3d9.h")
    endif()
    add_library(SIRD3D9 MODULE SIRD3D9Driver.cpp SIRD3D9Native.cpp SIRD3D9Formats.cpp SIRD3D9Targets.cpp)
    target_compile_definitions(SIRD3D9 PRIVATE SIR_D3D9_EXPORTS=1)
    target_include_directories(SIRD3D9 PRIVATE "${PROJECT_SOURCE_DIR}/Source" "${PROJECT_SOURCE_DIR}/Build/CMake")
    target_link_libraries(SIRD3D9 PRIVATE SIR::Graphics SIRD3D9Portable d3d9 user32)
    target_compile_features(SIRD3D9 PRIVATE cxx_std_17)
    sir_configure_owned_target(SIRD3D9)
    if(SIR_BUILD_TESTS)
        add_executable(SIRD3D9NativeSmoke "${PROJECT_SOURCE_DIR}/Tests/Renderer/D3D9/SIRD3D9NativeSmoke.cpp" SIRD3D9Native.cpp SIRD3D9Formats.cpp SIRD3D9Targets.cpp)
        target_include_directories(SIRD3D9NativeSmoke PRIVATE "${PROJECT_SOURCE_DIR}/Source" "${PROJECT_SOURCE_DIR}/Build/CMake")
        target_compile_features(SIRD3D9NativeSmoke PRIVATE cxx_std_17)
        target_link_libraries(SIRD3D9NativeSmoke PRIVATE SIRD3D9Portable SIR::Graphics d3d9 user32)
        sir_configure_owned_target(SIRD3D9NativeSmoke)
        add_test(NAME renderer.d3d9.native-smoke COMMAND SIRD3D9NativeSmoke)
        add_test(NAME renderer.d3d9ex.native-smoke COMMAND SIRD3D9NativeSmoke --ex)
    endif()
endif()

cmake_minimum_required(VERSION 4.4.5)
include(Build/CMake/SIRBuildTreeGuard.cmake)
project(SIREngine VERSION 0.1.0 LANGUAGES C CXX)

# Primary C++23; C++17 only for toolchains lacking the C++23 language mode.
# SIR_CXX_STANDARD=17 exercises the fallback during qualification.
set(SIR_CXX_STANDARD "AUTO" CACHE STRING "SIREngine C++ mode: AUTO, 23, or 17")
set_property(CACHE SIR_CXX_STANDARD PROPERTY STRINGS AUTO 23 17)
if(SIR_CXX_STANDARD STREQUAL "AUTO")
    if("cxx_std_23" IN_LIST CMAKE_CXX_COMPILE_FEATURES)
        set(SIR_SELECTED_CXX_STANDARD 23)
    elseif("cxx_std_17" IN_LIST CMAKE_CXX_COMPILE_FEATURES)
        set(SIR_SELECTED_CXX_STANDARD 17)
    else()
        message(FATAL_ERROR "SIREngine requires at least C++17")
    endif()
elseif(SIR_CXX_STANDARD STREQUAL "23" OR SIR_CXX_STANDARD STREQUAL "17")
    set(SIR_SELECTED_CXX_STANDARD "${SIR_CXX_STANDARD}")
else()
    message(FATAL_ERROR "SIR_CXX_STANDARD must be AUTO, 23, or 17")
endif()
message(STATUS "SIREngine C++ standard: C++${SIR_SELECTED_CXX_STANDARD}")
set(CMAKE_CXX_STANDARD ${SIR_SELECTED_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(SIR_BUILD_TESTS "Build and register native unit tests" ON)
option(SIR_BUILD_BENCHMARKS "Build native CPU microbenchmarks" OFF)
option(SIR_WITH_GOOGLETEST "Enable GoogleTest for engine unit tests" OFF)
option(SIR_WITH_GOOGLE_BENCHMARK "Enable Google Benchmark for profiling" OFF)
option(SIR_WITH_EASTL "Build pinned EASTL for future ported systems" OFF)
option(SIR_WITH_FMOD "Resolve the external FMOD Studio API" OFF)
option(SIR_WITH_WWISE "Resolve the external Wwise SDK" OFF)
option(SIR_WITH_OPENAL "Resolve a system OpenAL implementation" OFF)
option(SIR_WITH_OPENAL_SOFT "Build pinned OpenAL Soft" OFF)
option(SIR_WITH_MESHOPTIMIZER "Build pinned meshoptimizer" OFF)
option(SIR_WITH_MONO "Resolve Mono embedding SDK for C# scripting" OFF)
option(SIR_WITH_VULKAN "Build native Vulkan 1.4 driver" OFF)
option(SIR_WITH_IMGUI_VULKAN "Build pinned Dear ImGui with the native Vulkan backend" OFF)
option(SIR_WITH_IMGUI_OPENGL "Build pinned Dear ImGui with the OpenGL 3 backend" OFF)
option(SIR_WITH_D3D12 "Build native Direct3D 12 driver (Windows only)" OFF)
option(SIR_WITH_OPENGL "Build the OpenGL 3.3 Core programmable RHI" OFF)

include(Build/CMake/SIRBuildOptions.cmake)
# libstdc++ selects its TBB execution backend when TBB headers are installed.
# The shared PCH includes <execution>, including in standalone test targets.
find_package(TBB QUIET COMPONENTS tbb)
if(TARGET TBB::tbb)
    link_libraries(TBB::tbb)
endif()
include(Build/CMake/SIRDependencies.cmake)
include(Build/CMake/SIRAudio.cmake)
include(Build/CMake/SIRMono.cmake)

# CTest must be enabled before configuring source-owned test targets.
if(SIR_BUILD_TESTS)
    enable_testing()
endif()

add_subdirectory(Source/Core)
add_subdirectory(Source/Resources)
add_subdirectory(Source/Streaming)
add_subdirectory(Source/Mesh)
add_subdirectory(Source/Renderer)
add_subdirectory(Source/Animation)
add_subdirectory(Source/Audio)
add_subdirectory(Source/Scripting)
add_subdirectory(Source/Terrain)
add_subdirectory(Source/World)
add_subdirectory(Source/UI)
add_subdirectory(Source/Editor)

add_executable(SIREngine Source/Main.cpp)
target_link_libraries(SIREngine PRIVATE SIR::Core SIR::Scripting)
target_compile_definitions(SIREngine PRIVATE SIR_VERSION="${PROJECT_VERSION}")
sir_configure_owned_target(SIREngine)
add_subdirectory(Tools)

if(SIR_BUILD_TESTS)
    add_subdirectory(Tests)
    add_subdirectory(Tests/Dependencies)
    include(Build/CMake/SIRTestRegistration.cmake)
    foreach(sirSubsystem Core Resources Streaming Mesh Renderer Animation Audio Scripting Terrain World UI Editor)
        sir_classify_tests("${PROJECT_SOURCE_DIR}/Source/${sirSubsystem}")
    endforeach()
    get_property(sirDependencyTests DIRECTORY "${PROJECT_SOURCE_DIR}/Tests/Dependencies" PROPERTY TESTS)
    if(sirDependencyTests)
        sir_classify_tests("${PROJECT_SOURCE_DIR}/Tests/Dependencies")
    endif()
endif()

if(SIR_BUILD_BENCHMARKS)
    if(NOT SIR_WITH_GOOGLE_BENCHMARK)
        message(FATAL_ERROR "SIR_BUILD_BENCHMARKS requires SIR_WITH_GOOGLE_BENCHMARK=ON")
    endif()
    add_subdirectory(Tests/Benchmarks)
endif()

include(Build/CMake/SIRStandards.cmake)

include(GNUInstallDirs)
install(TARGETS SIREngine RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
set(CPACK_PACKAGE_NAME "SIREngine")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_GENERATOR "ZIP")
include(CPack)

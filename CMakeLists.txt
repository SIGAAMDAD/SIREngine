cmake_minimum_required(VERSION 4.4.5)
include(Build/CMake/SIRBuildTreeGuard.cmake)
project(SIREngine VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(SIR_BUILD_TESTS "Build and register native unit tests" ON)
option(SIR_BUILD_BENCHMARKS "Build native CPU microbenchmarks" OFF)
option(SIR_WITH_GOOGLETEST "Enable GoogleTest for engine unit tests" OFF)
option(SIR_WITH_GOOGLE_BENCHMARK "Enable Google Benchmark for profiling" OFF)
option(SIR_WITH_EASTL "Build pinned EASTL for future ported systems" OFF)
option(SIR_WITH_FMOD "Resolve the external FMOD Studio API" OFF)
option(SIR_WITH_WWISE "Resolve the external Wwise SDK" OFF)
option(SIR_WITH_OPENAL "Resolve a system OpenAL implementation" OFF)
option(SIR_WITH_OPENAL_SOFT "Build pinned OpenAL Soft" OFF)
option(SIR_WITH_MESHOPTIMIZER "Build pinned meshoptimizer" OFF)
option(SIR_WITH_MONO "Resolve Mono embedding SDK for C# scripting" OFF)
option(SIR_WITH_VULKAN "Build native Vulkan 1.4 driver" OFF)
option(SIR_WITH_D3D12 "Build native Direct3D 12 driver (Windows only)" OFF)
option(SIR_WITH_OPENGL "Build native OpenGL 4.6 driver" OFF)


# Platform, driver and offline-cooker capabilities are configured once here.
option(SIR_WITH_GAMEINPUT "Enable native Windows GameInput controller history" ON)
option(SIR_REQUIRE_GAMEINPUT_SDK "Require GameInput SDK during configure (CI acceptance)" OFF)
option(SIR_WITH_WAYLAND_WINDOW "Build native Linux Wayland xdg-shell window driver" ON)
option(SIR_WITH_WAYLAND_INPUT "Build first-party Wayland seat input bridge" ON)
option(SIR_REQUIRE_WAYLAND_INPUT "Require native Wayland keyboard/pointer input at configure" OFF)
option(SIR_REQUIRE_WAYLAND_WINDOW "Require native Wayland window driver at configure" OFF)
option(SIR_WITH_X11_WINDOW "Build the native X11 window driver when X11 is available" ON)
option(SIR_WITH_X11_INPUT "Build native X11 input bridge when X11 is available" ON)
option(SIR_WITH_D3D9 "Build the native Direct3D 9/9Ex backend" OFF)
option(SIR_WITH_D3D10 "Build the native Windows Direct3D 10 backend" OFF)
option(SIR_WITH_D3D11 "Build the native Windows Direct3D 11 backend" OFF)
set(SIR_SHADER_DXC_EXECUTABLE "" CACHE FILEPATH "Pinned offline DXC executable for shader acceptance")
set(SIR_SHADER_FXC_EXECUTABLE "" CACHE FILEPATH "Pinned offline FXC executable for D3D9 shader acceptance")

include(Build/CMake/SIRBuildOptions.cmake)
include(Build/CMake/SIRDependencies.cmake)
include(Build/CMake/SIRAudio.cmake)
include(Build/CMake/SIRMono.cmake)

# CTest must be enabled before configuring source-owned test targets.
if(SIR_BUILD_TESTS)
    enable_testing()
endif()

# Each major system owns exactly one build declaration.
add_subdirectory(Source/Core)
add_subdirectory(Source/Resources)
add_subdirectory(Source/Renderer)
add_subdirectory(Source/Terrain)
add_subdirectory(Source/Mesh)
add_subdirectory(Source/Streaming)
add_subdirectory(Source/World)
add_subdirectory(Source/UI)
add_subdirectory(Source/Editor)
add_subdirectory(Source/Audio)
add_subdirectory(Source/Scripting)
add_subdirectory(Source/Animation)

add_executable(SIREngine ${PROJECT_SOURCE_DIR}/Source/Main.cpp)
target_link_libraries(SIREngine PRIVATE SIR::Core SIR::Resources SIR::Graphics SIR::Scripting)
target_compile_definitions(SIREngine PRIVATE SIR_VERSION="${PROJECT_VERSION}")

sir_configure_owned_target(SIREngine)


# Build all production tools before registering test/qualification targets.
add_subdirectory(Tools)
if(SIR_BUILD_TESTS OR SIR_BUILD_BENCHMARKS)
    if(SIR_BUILD_BENCHMARKS AND NOT SIR_WITH_GOOGLE_BENCHMARK)
        message(FATAL_ERROR "SIR_BUILD_BENCHMARKS requires SIR_WITH_GOOGLE_BENCHMARK=ON")
    endif()
    add_subdirectory(Tests)
endif()


include(GNUInstallDirs)
install(TARGETS SIREngine RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}" COMPONENT runtime)
install(TARGETS SIREditor RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}" COMPONENT editor)
install(TARGETS SIRRegistrationCooker SIRShaderCooker RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}" COMPONENT tools)
if(TARGET sir-geometry-cooker)
    install(TARGETS sir-geometry-cooker RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}" COMPONENT tools)
endif()
set(CPACK_PACKAGE_NAME "SIREngine")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_GENERATOR "ZIP")
include(CPack)

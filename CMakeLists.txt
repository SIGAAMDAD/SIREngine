cmake_minimum_required(VERSION 4.4.5)
project(SIREngine VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(SIR_BUILD_TESTS "Build and register native unit tests" ON)
option(SIR_BUILD_BENCHMARKS "Build native CPU microbenchmarks" OFF)
option(SIR_WITH_GOOGLETEST "Enable GoogleTest for engine unit tests" OFF)
option(SIR_WITH_GOOGLE_BENCHMARK "Enable Google Benchmark for profiling" OFF)
option(SIR_WITH_EASTL "Build pinned EASTL for future ported systems" OFF)
option(SIR_WITH_FMOD "Resolve the external FMOD Studio API" OFF)
option(SIR_WITH_WWISE "Resolve the external Wwise SDK" OFF)
option(SIR_WITH_OPENAL "Resolve a system OpenAL implementation" OFF)
option(SIR_WITH_OPENAL_SOFT "Build pinned OpenAL Soft" OFF)
option(SIR_WITH_MESHOPTIMIZER "Build pinned meshoptimizer" OFF)
option(SIR_WITH_MONO "Resolve Mono embedding SDK for C# scripting" OFF)
option(SIR_WITH_VULKAN "Build native Vulkan 1.4 driver" OFF)

include(Build/CMake/SIRBuildOptions.cmake)
include(Build/CMake/SIRDependencies.cmake)
include(Build/CMake/SIRAudio.cmake)
include(Build/CMake/SIRMono.cmake)
add_subdirectory(Source)

if(SIR_BUILD_TESTS)
    enable_testing()
    add_subdirectory(Tests/Unit)
    add_subdirectory(Tests/Dependencies)
endif()

if(SIR_BUILD_BENCHMARKS)
    if(NOT SIR_WITH_GOOGLE_BENCHMARK)
        message(FATAL_ERROR "SIR_BUILD_BENCHMARKS requires SIR_WITH_GOOGLE_BENCHMARK=ON")
    endif()
    add_subdirectory(Tests/Benchmarks)
endif()

include(GNUInstallDirs)
install(TARGETS SIREngine RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
set(CPACK_PACKAGE_NAME "SIREngine")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_GENERATOR "ZIP")
include(CPack)

add_library(SIRResources STATIC Private/SIRResourceGraph.cpp)
target_include_directories(SIRResources PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Public")
target_compile_features(SIRResources PUBLIC cxx_std_17)
find_package(Threads REQUIRED)
target_link_libraries(SIRResources PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(SIRResources PRIVATE /W4)
else()
    target_compile_options(SIRResources PRIVATE -Wall -Wextra -Wpedantic)
endif()
sir_configure_owned_target(SIRResources)

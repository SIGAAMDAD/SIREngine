cmake_minimum_required(VERSION 3.20)
project(SIRAnimationCookQualification LANGUAGES CXX)
include(CTest)
add_library(SIRAnimationCook STATIC
    ../../../Source/Animation/Cooked/SIRAnimationCooked.cpp
    ../../../Tools/SceneCooker/Animation/SIRAnimationCooker.cpp)
target_compile_features(SIRAnimationCook PUBLIC cxx_std_17)
target_include_directories(SIRAnimationCook PUBLIC ../../../Source ../../../Build/CMake ../../../Tools)
add_executable(SIRAnimationCookTests SIRAnimationCookTests.cpp)
target_link_libraries(SIRAnimationCookTests PRIVATE SIRAnimationCook)
add_test(NAME ANM01_Cook COMMAND SIRAnimationCookTests)
if(TARGET gtest_main)
    set(SIR_ANM01_GTEST_TARGET gtest_main)
else()
    find_package(GTest QUIET)
    if(TARGET GTest::gtest_main)
        set(SIR_ANM01_GTEST_TARGET GTest::gtest_main)
    endif()
endif()
if(SIR_ANM01_GTEST_TARGET)
    add_executable(SIRAnimationCookGoogleTests SIRAnimationCookGoogleTests.cpp)
    target_link_libraries(SIRAnimationCookGoogleTests PRIVATE SIRAnimationCook ${SIR_ANM01_GTEST_TARGET})
    include(GoogleTest)
    gtest_discover_tests(SIRAnimationCookGoogleTests TEST_PREFIX "animation.cook.")
endif()
